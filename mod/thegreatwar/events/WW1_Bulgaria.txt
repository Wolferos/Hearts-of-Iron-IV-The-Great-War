add_namespace = ww1_bulgaria
# When Bulgaria is asked to join the Central Powers 
#____________________________________________________________
country_event = {
	id = ww1_bulgaria.1
	title = ww1_bulgaria.1.t
	desc = ww1_bulgaria.1.d
	picture = ww1_bulgaria_1
	fire_only_once = yes
	is_triggered_only = yes
	option = {
		name = ww1_bulgaria.1.a
		ai_chance = {
			base = 100
		}
		GER = {
			send_equipment = {
				target = BUL
				type = infantry_equipment_1
				amount = 800
			}
			add_to_faction = BUL
		}
		hidden_effect = {
			BUL = {
				remove_opinion_modifier = {
					target = TUR
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
			}
		}
	}
	option = {
		name = ww1_bulgaria.1.b
		ai_chance = {
			base = 0
		}
	}
}

# Albania Partition
country_event = {
	id = ww1_bulgaria.5
	title = ww1_bulgaria.5.t
	desc = ww1_bulgaria.5.d

	is_triggered_only = yes

	option = {
		name = ww1_bulgaria.5.a  ## accept demands
		ai_chance = {
			base = 100
		}
		GRE = { transfer_state = 803 }
		SER = {
			annex_country = {
				target = ALB
				transfer_troops = no
			}
		}
	}

	option = {
		name = ww1_bulgaria.5.b  ## refuse demands
		ai_chance = {
			base = 0
		}
		GRE = {
			declare_war_on = {
				target = ALB
				type = annex_everything
			}
		}
		SER = {
			declare_war_on = {
				target = ALB
				type = annex_everything
			}
		}
		BUL = {
			declare_war_on = {
				target = ALB
				type = annex_everything
			}
		}
	}
}
