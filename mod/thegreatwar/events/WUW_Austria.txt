			}

		}

				is_fully_controlled_by = ROOT
			}
		}		
	}

country_event = { #Austria informed of Italy backing down
	id = AUS_political_events.105
	title = AUS_political_events.105.t
	desc = AUS_political_events.105.desc
	picture = GFX_report_event_generic_riot
	is_triggered_only = yes

	option = { #we better back down
		name = AUS_political_events.105.a
		ai_chance = {
			base = 10
		}

		GER = {
			country_event = { hours = 2 id = germany.4 }
		}
		custom_effect_tooltip = GAME_OVER_TT

		AUS = {
			clr_country_flag = AUS_antischluss_active_flag
		}
	}

	option = { #keep fighting!
		name = AUS_political_events.105.b
		ai_chance = {
			base = 10
		}

		effect_tooltip = {
			GER = {
				add_state_claim = 4
				add_state_claim = 152
				add_state_claim = 153
				add_state_claim = 848
				create_wargoal = {
					type = take_state_focus
					target = AUS
					generator = { 4 152 153 848 }
				}
				
			}

			news_event = {
				id = AUS_news_events.1
				days = 3 
			}

			AUS = {
				clr_country_flag = AUS_antischluss_active_flag
				IF = {
					limit = { 
						NOT = { 
							has_country_flag = AUS_seyss_inquart_can_be_fired_flag 
						} 
					}
					set_can_be_fired_in_advisor_role = {
						character = AUS_arthur_seyss-inquart
						value = yes
					}
					set_country_flag = AUS_seyss_inquart_can_be_fired_flag
				}
			}
		}
		
		GER = {
			add_opinion_modifier = { target = ROOT modifier = GER_anschluss_rejected }
			country_event = { days = 1 id = germany.29 }
			if = { limit = { is_in_faction_with = AUS }
				remove_from_faction = AUS
			}
		}
		
	}
}
