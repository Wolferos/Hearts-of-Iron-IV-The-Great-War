	title = ww1_raj.5.t
	desc = ww1_raj.5.d
	id = ww1_raj.5

	is_triggered_only = yes
	
	option = {
		name = ww1_raj.5.a	#Leave Baluchistan
		trigger = { owns_state = 444 }
		every_owned_state = {
			limit = { is_core_of = PAK }
			PAK = { transfer_state = PREV }
		}
		if = {
			limit = { has_idea = RAJ_cooperation_with_muslim_community }
			remove_ideas = RAJ_cooperation_with_muslim_community
		}
		if = {
			limit = { has_idea = RAJ_muslim_community }
			remove_ideas = RAJ_muslim_community
		}
		ai_chance = { base = 1 }
	}
	option = {
		name = ww1_raj.5.b	#Confederation
		trigger = { has_idea = RAJ_cooperation_with_muslim_community }
		swap_ideas = {
			remove_idea = RAJ_cooperation_with_muslim_community
			add_idea = RAJ_muslim_community_happy
		}
		ai_chance = { base = 10 }
	}
	option = {
		name = ww1_raj.5.c	#Expand representation
		trigger = { has_idea = RAJ_muslim_community }
		add_political_power = -200
		swap_ideas = {
			remove_idea = RAJ_muslim_community
			add_idea = RAJ_muslim_community_discontent
		}
		ai_chance = { base = 10 }
	}
}

# Afghan bands raid us
country_event = {
	id = ww1_raj.11
	title = ww1_raj.11.t
	desc = ww1_raj.11.d

	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			set_temp_variable = {
				var_name = raid_victims
			}
			randomize_temp_variable = { 
				var = raid_victims
				distribution = binomial 
				min = -1000
				max = -150
			}
		}
	}
	
	option = {
		name = ww1_raj.11.a
		add_manpower = var:raid_victims
		random_controlled_state = {
			limit = { is_core_of = PAK }
			damage_building = {
				type = infrastructure
				damage = 1
			}
			if = {
				limit = { NOT = { is_core_of = RAJ } resistance < 50 }
				add_resistance = 10
			}
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_completed_focus = RAJ_establish_the_order_on_durand_line }
					is_neighbor_of = AFG
					NOT = { has_war_with = AFG }
				}
				random_list = {
					25 = { 
						country_event = { id = ww1_raj.11 days = 60 } 
					}
					25 = { 
						country_event = { id = ww1_raj.11 days = 90 } 
					}
					25 = { 
						country_event = { id = ww1_raj.11 days = 120 } 
					}
					25 = { 
						country_event = { id = ww1_raj.11 days = 180 } 
					}
				}
			}
		}
	}
}
