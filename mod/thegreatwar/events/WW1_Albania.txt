#-> Assembly of Vlorë <><>
country_event = {
	id = ww1_albania.4
	title = ww1_albania.4.t
	desc = ww1_albania.4.desc
	picture = GFX_report_event_SWI_assembly_hall

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = ww1_albania.4.a
		TUR = {
			set_country_flag = form_albania
			set_country_flag = alb_peace
		}
		if = {
			limit = {
				TUR = {
					controls_state = 766
				}
			}
			transfer_state = 766
		}
		if = {
			limit = {
				TUR = {
					controls_state = 44
				}
			}
			transfer_state = 44
		}
		hidden_effect = {
			44 = {
				if = {
					limit = {
						is_border_conflict = yes
					}
					set_border_war = no
				}
			}
			766 = {
				if = {
					limit = {
						is_border_conflict = yes
					}
					set_border_war = no
				}
			}
		}
		set_capital = {
			state = 44
		}
	}
}
