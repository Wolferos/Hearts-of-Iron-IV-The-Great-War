country_event = {
	id =  int_germany.4
	title = int_germany.4.t
	desc = int_germany.4.d
	picture = GFX_report_event_gathering_protest
	
	fire_only_once = yes

	trigger = {
		tag = GER
		has_global_flag = kis_treaty_of_versailles
		date > 1923.11.8
		date < 1923.11.9
		has_war = no
		is_subject = no
		NOT = {
			has_government = fascism
		}
		ITA = {
			has_completed_focus = ITA_interwar_march_on_rome
		}
	}

	option = {
		name = int_germany.4.a
		ai_chance = { factor = 50 }

		set_global_flag = interwar_hitler_putsch_success

		hidden_effect = {
			GER = {
				add_popularity = {
					ideology = fascism
					popularity = 0.72
				}
				set_politics = {
					ruling_party = fascism
					elections_allowed = no
				}
			}
		}
	}
	option = {
		name = int_germany.4.b
		ai_chance = { factor = 50 }

		set_global_flag = interwar_hitler_putsch_fail
	}
}
