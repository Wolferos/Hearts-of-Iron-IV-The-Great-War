add_namespace = ww1_mexico

# Start of the Mexican Revolution (In Chihuahua State on November 10, 1910)
#____________________________________________________________
country_event = {
	id = ww1_mexico.1
	title = ww1_mexico.1.t
	desc = ww1_mexico.1.d
	picture = EVENT_hoi4tgw_mexican_revolution

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { has_global_flag = FLAG_MEX_adios_porfirio_diaz }
		has_defensive_war = no
		NOT = { has_government = democratic }
		NOT = { has_completed_focus = MEX_ww1_supress_the_rebels }
	}

	immediate = {
		hidden_effect = {
			set_global_flag = SUPER_GLOBAL_Mexican_Revolution_1910
			news_event = { id = ww1_mexico.2 hours = 6 }
		}
	}

	option = { # Support Revolution (Play as Francisco I. Madero)
		name = ww1_mexico.1.a
		hidden_effect = {
			MEX = {
				set_capital = { state = 482 } # Chihuahua State
				start_civil_war = {
					ruling_party = democratic
					ideology = neutrality
					size = 0.5
					capital = 277
					states = { 277 477 479 478 485 }
				}
				random_other_country = {
					limit = {
						original_tag = MEX
						has_government = neutrality
					}
					load_oob = "MEX_1910"
					set_equipment_fraction = 0.4
					set_stability = 0.35
					save_global_event_target_as = porfirio_loyalist_mexico
					every_character = {
						limit = {
							has_character_flag = MEX_porfirio_loyalist
						}
						set_nationality = event_target:porfirio_loyalist_mexico
					}
					add_manpower = -80000
				}
				add_manpower = 200000
				load_oob = "Mexican_Revolution_1910"
				set_equipment_fraction = 0.9
				set_stability = 0.8
				add_popularity = {
					ideology = democratic
					popularity = 0.7
				}
			}
		}
	}

	option = { # Fight Revolution (Stay with Diaz)
		name = ww1_mexico.1.b
		trigger = {
			is_ai = no
		}
		hidden_effect = {
			MEX = {
				start_civil_war = {
					ideology = democratic
					size = 0.5
					capital = 482
					states = { 482 483 481 484 474 475 480 476  }
				}
				random_other_country = {
					limit = {
						original_tag = MEX
						has_government = democratic
					}
					load_oob = "Mexican_Revolution_1910"
					add_manpower = -8000
					save_global_event_target_as = revolutionary_mexico
					every_character = {
						limit = {
							has_character_flag = MEX_revolutionary
						}
						set_nationality = event_target:revolutionary_mexico
					}
					set_equipment_fraction = 0.9
					set_stability = 0.9
				}

				set_politics = {
					ruling_party = neutrality
					elections_allowed = yes
				}
				add_popularity = {
					ideology = democratic
					popularity = 0.05
				}
				add_popularity = {
					ideology = neutrality
					popularity = 1
				}
				add_manpower = 20000

				load_oob = "MEX_1910"
				set_equipment_fraction = 0.9
				set_stability = 0.5
			}
		}
	}
}

					name = "Victoriano Huerta"
