QIN_decrease_debt_burden = {
	if = {
		limit = {
			has_idea = QIN_massive_debt_burden_1
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_1
			add_idea = QIN_massive_debt_burden_2
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_2
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_2
			add_idea = QIN_massive_debt_burden_3
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_3
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_3
			add_idea = QIN_massive_debt_burden_4
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_4
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_4
			add_idea = QIN_massive_debt_burden_5
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_5
		}
		remove_ideas = QIN_massive_debt_burden_5
	}
	else = {
		add_political_power = 30
	}
}

QIN_increase_debt_burden = {
	if = {
		limit = {
			has_idea = QIN_massive_debt_burden_5
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_5
			add_idea = QIN_massive_debt_burden_4
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_4
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_4
			add_idea = QIN_massive_debt_burden_3
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_3
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_3
			add_idea = QIN_massive_debt_burden_2
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_2
		}
		swap_ideas = {
			remove_idea = QIN_massive_debt_burden_2
			add_idea = QIN_massive_debt_burden_1
		}
	}
	else_if = {
		limit = {
			has_idea = QIN_massive_debt_burden_1
		}
		add_stability = -0.2
	}
	else = {
		add_ideas = QIN_massive_debt_burden_5
	}
}
