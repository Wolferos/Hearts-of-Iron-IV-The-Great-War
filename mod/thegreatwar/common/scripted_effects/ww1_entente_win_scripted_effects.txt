# Liberate Brest-Litovsk Puppets Script
ww1_versailles_setup_liberation_brest_litovsk_puppets = {
	if = {
		limit = {
			any_country_with_original_tag_of = {
				target = { RUS SOV }
				has_country_flag = great_war_loser
			}
		}
		if = {
			limit = {
				has_subject = LIT
			}
			set_autonomy = {
				target = LIT
				autonomy_state = autonomy_free
			}
		}
		if = {
			limit = {
				has_subject = UBD
			}
			set_autonomy = {
				target = UBD
				autonomy_state = autonomy_free
			}
		}
		if = {
			limit = {
				has_subject = POL
			}
			set_autonomy = {
				target = POL
				autonomy_state = autonomy_free
			}
		}
		if = {
			limit = {
				has_subject = UKR
			}
			set_autonomy = {
				target = UKR
				autonomy_state = autonomy_free
			}
		}
		if = {
			limit = {
				has_subject = BLR
			}
			set_autonomy = {

# Asir Script
ww1_versailles_setup_asir = {
	if = {
		limit = {
			any_enemy_country = {
				owns_state = 856
			}
		}
		if = {
			limit = {
				ASR = {
					exists = no
				}
			}
			set_global_flag = ASR_not_existing
		}
		if = {
			limit = {
				ENT = {
					has_war_with = ASR
				}
			}
			every_country_with_original_tag = {
				original_tag_to_check = TUR
				limit = {
					has_subject = ASR
				}
				set_autonomy = {
					target = ASR
					autonomy_state = autonomy_free
				}
			}
			SAU = {
				if = {
					limit = {
						NOT = {
							is_in_faction_with = CPL
						}
					}
					add_state_core = 856
				}
			}
			if = {
				limit = {
					SAU = {
						is_in_faction_with = ENT
					}
				}
				SAU = {
					transfer_state = 856
				}
			}
		}
		if = {
			limit = {
				country_exists = ASR
				has_global_flag = ASR_not_existing
			}
			ASR = {
				load_oob = "ASR_1914"
			}
		}
		clr_global_flag = ASR_not_existing
	}
}
