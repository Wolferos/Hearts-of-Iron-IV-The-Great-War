
on_actions = {

	on_release_as_free = {
		effect = {
			# Marijan Varešanin Transfer to Croatia
			if = {
				limit = {
					country_exists = CRO
					NOT = {
						CRO = {
							has_character = BOS_marijan_varesanin
						}
					}
				}
				BOS_marijan_varesanin = {
					set_nationality = CRO
				}
			}
		}

		effect = {
			# Franz Albert Seyn Transfer to Russia
			if = {
				limit = {
					ROOT = {
						tag = FIN
					}
					country_exists = RUS
					NOT = {
						RUS = {
							has_character = RUS_franz_albert_seyn
						}
					}
				}
				RUS_franz_albert_seyn = {
					set_nationality = RUS
					remove_country_leader_role = {
						ideology = despotism
					}
				}
			}
		}

		effect = {
			# Cosmetic Tag Removal
			if = {
				limit = {
					OR = {
						has_cosmetic_tag = FIN_RUS_neutrality_subject
					}
				}
				drop_cosmetic_tag = yes
			}
		}
	}

				limit = {
					original_tag = ITA

						OR = {
							controls_state = 826
							controls_state = 273
							controls_state = 662
							controls_state = 663
						}
						NOT = {
							has_completed_focus = ITA_ww1_italian_askari
						}
						has_war_with_major = yes
						NOT = {
							is_in_faction_with = TUR
						}
					}
				}
				country_event = { id = ww1_libya.1 random_days = 30 }
			}
		}
	}

	on_peaceconference_ended = {

		# Italy Annexes Senussi
		effect = {
			if = {
				limit = {
					NOT = {
						has_global_flag = hoi4tgw_setting_scripted_peace_conferences_disabled
					}

						original_tag = ITA
					}
					FROM = {
						original_tag = LBA
					}
				}
				ROOT = {
					annex_country = {
						target = FROM
					}
				}
			}
		}

		# Russian Civil War End
		effect = {
			if = {
				limit = {
					NOT = {
						has_global_flag = hoi4tgw_setting_scripted_peace_conferences_disabled
					}
